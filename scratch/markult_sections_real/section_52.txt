// EXISTING — already in qrCheckInService.js
const generateCheckInToken = async (prisma, { reservationId, customerId }) => {
    if (!reservationId) throw new Error('reservationId is required.');
    if (!customerId) throw new Error('customerId is required.');

    const reservation = await prisma.reservation.findUnique({
        where: { id: reservationId },
        include: { escrow: true, businessProfile: true },
    });
    if (!reservation) throw new Error('Reservation not found.');
    if (reservation.customerId !== customerId)
        throw new Error('Not authorized — reservation does not belong to this customer.');
    if (reservation.status !== 'CONFIRMED')
        throw new Error(`Reservation is ${reservation.status}, must be CONFIRMED.`);
    if (!reservation.escrow || reservation.escrow.status !== 'FUNDED')
        throw new Error('Escrow is not funded. Cannot generate check-in token.');

    const azaman = await prisma.azamanIdentity.findFirst({
        where: { userId: customerId, status: 'ACTIVE' },
    });
    if (!azaman) throw new Error('No active AZM identity for customer.');

    const payload = {
        rid: reservationId,
        cid: customerId,
        aid: azaman.azamanId,
        ts: Date.now(),
    };
    const token = _signToken(payload, TOKEN_TTL_MINUTES);
    return {
        token,
        qrData: JSON.stringify({ t: token, v: 1 }),
        expiresIn: TOKEN_TTL_MINUTES * 60,
        businessName: reservation.businessProfile?.businessName,
        reservationDate: reservation.startDatetime,
    };
};

const generateTransitCheckInToken = async (prisma, { transitBookingId, customerId }) => {
    if (!transitBookingId) throw new Error('transitBookingId is required.');
    if (!customerId) throw new Error('customerId is required.');

    const booking = await prisma.transitBooking.findUnique({
        where: { id: transitBookingId },
        include: { trip: { include: { businessProfile: true } }, escrow: true },
    });
    if (!booking) throw new Error('Transit booking not found.');
    if (booking.customerId !== customerId)
        throw new Error('Not authorized — booking does not belong to this customer.');
    if (booking.status !== 'CONFIRMED')
        throw new Error(`Booking is ${booking.status}, must be CONFIRMED.`);
    if (!booking.escrow || booking.escrow.status !== 'FUNDED')
        throw new Error('Escrow is not funded. Cannot generate boarding token.');

    const azaman = await prisma.azamanIdentity.findFirst({
        where: { userId: customerId, status: 'ACTIVE' },
    });
    if (!azaman) throw new Error('No active AZM identity for customer.');

    const payload = {
        rid: transitBookingId, // reuse "rid" field for booking ID
        cid: customerId,
        aid: azaman.azamanId,
        ts: Date.now(),
        type: 'TRANSIT', // distinguish from reservation
        seats: booking.seatNumbers,
    };
    const token = _signToken(payload, TOKEN_TTL_MINUTES);
    return {
        token,
        qrData: JSON.stringify({ t: token, v: 1 }),
        expiresIn: TOKEN_TTL_MINUTES * 60,
        businessName: booking.trip?.businessProfile?.businessName,
        route: booking.trip?.routeLabel,
        departureTime: booking.trip?.scheduledAt,
        seatNumbers: booking.seatNumbers,
    };
};
