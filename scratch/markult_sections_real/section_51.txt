CRITICAL FIX: The existing getFeed() method only queries stories from friends. Business stories (created by reviewStoryService and the ad system) set businessProfileId but use the business owner userId — which may not be in the viewer friends list. This means business stories get CREATED but never RENDER in any follower feed.
The fix: ALSO include stories from businesses the viewer follows.

REPLACE with:
async getFeed(viewerId) {
    const friendIds = await this._getFriendIds(viewerId);

    // NEW: Also get IDs of businesses the viewer follows
    const followedBusinesses = await this.prisma.businessFollower.findMany({
        where: { followerId: viewerId },
        select: { businessProfileId: true },
    });
    const followedBizIds = followedBusinesses.map(f => f.businessProfileId);

    // Get the owner userIds of followed businesses (to include their stories)
    const followedBizOwners = followedBizIds.length > 0
        ? await this.prisma.businessProfile.findMany({
            where: { id: { in: followedBizIds } },
            select: { userId: true, id: true, businessName: true, logoUrl: true },
        })
        : [];
    const followedOwnerIds = followedBizOwners.map(b => b.userId);

    // Combine friend IDs and followed business owner IDs
    const allAuthorIds = [...new Set([...friendIds, ...followedOwnerIds])];

    const stories = await this.prisma.story.findMany({
        where: {
            expiresAt: { gt: new Date() },
            userId: { in: allAuthorIds },
        },
        orderBy: { createdAt: 'desc' },
        include: {
            user: { select: { id: true, username: true, profilePictureUrl: true } },
            businessProfile: {
                select: { id: true, businessName: true, logoUrl: true, businessType: true }
            },
        },
    });

    // Build a lookup map for business owners
    const bizOwnerMap = new Map();
    for (const b of followedBizOwners) {
        bizOwnerMap.set(b.userId, b);
    }

    // Group stories by author — separate friend stories from business stories
    const groups = [];
    const seenAuthors = new Set();

    for (const story of stories) {
        const authorId = story.userId;
        if (seenAuthors.has(authorId)) {
            const existing = groups.find(g => g.authorId === authorId);
            if (existing) existing.stories.push(story);
        } else {
            seenAuthors.add(authorId);
            const biz = story.businessProfile || bizOwnerMap.get(authorId);
            groups.push({
                authorId,
                authorName: biz ? biz.businessName : story.user.username,
                authorAvatar: biz ? biz.logoUrl : story.user.profilePictureUrl,
                isBusiness: !!biz,
                businessId: biz ? biz.id : null,
                businessType: biz ? biz.businessType : null,
                stories: [story],
            });
        }
    }

    return groups;
}

async getBusinessStories(businessProfileId) {
    const stories = await this.prisma.story.findMany({
        where: {
            businessProfileId,
            expiresAt: { gt: new Date() },
        },
        orderBy: { createdAt: 'desc' },
        include: {
            user: { select: { id: true, username: true, profilePictureUrl: true } },
        },
    });
    return stories;
}
