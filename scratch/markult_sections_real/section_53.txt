// =============================================================================
// sendTransitPreDepartureReminders — sends push + notification 1 hour before
// scheduled departure for confirmed transit bookings that haven't checked in.
// =============================================================================
const sendTransitPreDepartureReminders = async (prisma, notificationService) => {
    const now = new Date();
    const oneHourFromNow = new Date(now.getTime() + 60 * 60 * 1000);
    const fifteenMinsFromNow = new Date(now.getTime() + 15 * 60 * 1000);

    // Find bookings departing in the next hour (between 45-75 min from now)
    // that are confirmed and not yet reminded
    const upcoming = await prisma.transitBooking.findMany({
        where: {
            status: 'CONFIRMED',
            checkedInAt: null,
            preDepartureReminderSent: false,
            trip: {
                scheduledAt: { gte: fifteenMinsFromNow, lte: oneHourFromNow },
            },
        },
        include: {
            trip: { include: { businessProfile: true } },
            customer: { select: { id: true, username: true } },
        },
    });

    for (const booking of upcoming) {
        try {
            await notificationService.sendMarketplaceNotification({
                userId: booking.customerId,
                type: 'TRANSIT_PRE_DEPARTURE',
                title: 'Departing Soon',
                body: `Your trip ${booking.trip.routeLabel} departs at ${
                    booking.trip.scheduledAt.toLocaleTimeString()
                }. Don't forget to check in!`,
                deepLink: `/business-market/transit/${booking.tripId}/seats`,
                metadata: { bookingId: booking.id, tripId: booking.tripId },
            });

            // Mark as reminded
            await prisma.transitBooking.update({
                where: { id: booking.id },
                data: { preDepartureReminderSent: true },
            });
        } catch (e) {
            console.error(`[preDepartureReminder] booking ${booking.id}:`, e.message);
        }
    }
};
